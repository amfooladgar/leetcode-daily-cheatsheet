# Config / schema validation

# Loads .env into os.environ for local runs (see README.md's
# "cp .env.example .env" step and src/main.py's main()). Never overrides a

# at render time); Playwright drives the browser. See ARCHITECTURE.md "Why
# the renderer moved to HTML/CSS" for the rationale.

# Google Drive upload via OAuth 2.0 user credentials, not a service account
# -- see ARCHITECTURE.md "Why OAuth instead of a service account" (a service
# account has zero Drive storage quota and cannot upload file content into
# a personal, non-Workspace Drive folder).

google-auth>=2.35,<3

# Timezone-correct scheduling check (America/New_York, DST-safe)
